cmake_minimum_required(VERSION 3.18)
project(podcast_linux VERSION 0.1.0 LANGUAGES C)

include(GNUInstallDirs)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0>=3.24 gio-2.0>=2.66)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0>=1.18)
pkg_check_modules(FEEDS REQUIRED IMPORTED_TARGET libcurl>=7.85 libxml-2.0)

add_executable(podcast-linux src/main.c src/feed.c src/player.c tests/smoke.c)
target_include_directories(podcast-linux PRIVATE src)
set_target_properties(podcast-linux PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_compile_definitions(podcast-linux PRIVATE _POSIX_C_SOURCE=200809L _FILE_OFFSET_BITS=64)
target_compile_options(podcast-linux PRIVATE -Wall -Wextra -Wpedantic -Wformat=2)
target_link_libraries(podcast-linux PRIVATE PkgConfig::GTK3 PkgConfig::GST PkgConfig::FEEDS)
install(TARGETS podcast-linux RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

if(BUILD_TESTING)
    add_test(NAME linux-core COMMAND podcast-linux --smoke-test)
    set_tests_properties(linux-core PROPERTIES TIMEOUT 45 ENVIRONMENT "DISPLAY=;WAYLAND_DISPLAY=;G_DEBUG=fatal-warnings")
    find_program(XVFB_RUN xvfb-run)
    find_program(XAUTH xauth)
    find_program(XVFB Xvfb)
    if(XVFB_RUN AND XAUTH AND XVFB AND NOT CMAKE_CROSSCOMPILING)
        add_test(NAME linux-window COMMAND "${XVFB_RUN}" -a $<TARGET_FILE:podcast-linux> --window-smoke-test)
        set_tests_properties(linux-window PROPERTIES TIMEOUT 25 ENVIRONMENT "GDK_BACKEND=x11;G_DEBUG=fatal-warnings")
    endif()
endif()
